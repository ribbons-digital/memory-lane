# Codex Hook Adapter Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a context-budgeted Codex hook adapter that recalls relevant Memory Lane memories before prompts and autosaves durable memories after turns/tool use.

**Architecture:** Keep `@memory-lane/core` harness-neutral, add optional harness-neutral provenance to memory records, add `@memory-lane/lifecycle` for shared automation policy, and add `@memory-lane/codex-adapter` for Codex hook stdin/stdout mapping. The existing CLI creates `MemoryEngine` and delegates `memory-lane codex ...` commands to the Codex adapter.

**Tech Stack:** TypeScript ESM, Node.js stdlib, `node:test`, `tsx`, existing `@memory-lane/core` APIs, pnpm workspace packages.

Create `packages/lifecycle/src/injection.ts`:

```ts
import { containsLikelySecret, lexicalScore, normalizeMemoryText, type MemoryRecord, type RecallResult } from "@memory-lane/core"

export interface MemoryInjectionLimits {
  maxItems: number
  targetChars: number
  hardMaxChars: number
  absoluteMaxChars: number
}

export const CODEX_MEMORY_INJECTION_LIMITS: MemoryInjectionLimits = {
  maxItems: 6,
  targetChars: 1800,
  hardMaxChars: 3000,
  absoluteMaxChars: 6000,
}

const GENERIC_PROMPTS = new Set([
  "ok", "okay", "yes", "yep", "yeah", "sure", "sounds good",
  "go ahead", "continue", "proceed", "approved", "looks good", "thanks", "thank you",
])

const STOP_WORDS = new Set(["a","an","and","are","as","at","be","by","can","did","do","does","for","from","how","i","in","is","it","of","off","on","or","please","that","the","this","to","use","we","what","where","with","you"])

function normalizedPrompt(prompt: string): string {
  return prompt.toLowerCase().replace(/[^\p{L}\p{N}]+/gu, " ").trim().replace(/\s+/gu, " ")
}

function meaningfulTokens(prompt: string): string[] {
  const normalized = normalizedPrompt(prompt)
  return normalized ? normalized.split(" ").filter((token) => token && !STOP_WORDS.has(token)) : []
}

export function shouldSkipAutomaticInjection(prompt: string): boolean {
  const normalized = normalizedPrompt(prompt)
  if (!normalized) return true
  if (GENERIC_PROMPTS.has(normalized)) return true
  return meaningfulTokens(prompt).length === 0
}

function dedupeKey(text: string): string {
  return normalizeMemoryText(text).toLowerCase().replace(/\s+/gu, " ").trim()
}

function capLimits(options?: Partial<MemoryInjectionLimits>): MemoryInjectionLimits {
  const merged = { ...CODEX_MEMORY_INJECTION_LIMITS, ...options }
  const absolute = Math.max(1, merged.absoluteMaxChars)
  return {
    maxItems: Math.max(0, merged.maxItems),
    targetChars: Math.min(Math.max(0, merged.targetChars), absolute),
    hardMaxChars: Math.min(Math.max(0, merged.hardMaxChars), absolute),
    absoluteMaxChars: absolute,
  }
}

function truncateMemory(memory: MemoryRecord, remaining: number): MemoryRecord | undefined {
  if (remaining <= 20) return undefined
  if (memory.text.length <= remaining) return memory
  const slice = memory.text.slice(0, remaining - 1)
  const boundary = Math.max(slice.lastIndexOf(". "), slice.lastIndexOf("! "), slice.lastIndexOf("? "))
  const text = (boundary >= 40 ? slice.slice(0, boundary + 1) : slice.trimEnd()) + "…"
  return { ...memory, text }
}

export function selectMemoriesForInjection(
  prompt: string,
  result: RecallResult,
  options?: Partial<MemoryInjectionLimits>,
): MemoryRecord[] {
  if (shouldSkipAutomaticInjection(prompt)) return []
  const limits = capLimits(options)
  const requireLexical = !result.semantic.used || result.semantic.fallbackReason === "No semantic matches"
  const seen = new Set<string>()
  const selected: MemoryRecord[] = []
  let chars = 0

  for (const memory of result.memories) {
    if (selected.length >= limits.maxItems) break
    if (containsLikelySecret(memory.text)) continue
    const lex = lexicalScore(prompt, memory.text)
    if (requireLexical && lex <= 0) continue
    const key = dedupeKey(memory.text)
    if (!key || seen.has(key)) continue
    const remaining = limits.hardMaxChars - chars
    const candidate = truncateMemory(memory, remaining)
    if (!candidate) continue
    const nextChars = chars + candidate.text.length
    if (nextChars > limits.hardMaxChars) continue
    selected.push(candidate)
    seen.add(key)
    chars = nextChars
  }

  return selected
}

export function renderMemoryBlock(memories: MemoryRecord[]): string {
  if (!memories.length) return ""
  return ["## Relevant Memory", "", ...memories.map((memory) => `- ${memory.text}`)].join("\n")
}
```

```ts
import {
  containsLikelySecret,
  detectUserMemorySuggestion,
  inferCategory,
  inferMemoryKind,
  normalizeMemoryText,
  parseExplicitMemoryRequest,
} from "@memory-lane/core"
import type { MemoryCandidate, StopInput } from "./types.js"

function isQuestion(text: string): boolean {
  return /^(?:what|how|why|when|where|who|do|does|did|is|are|can|could|should)\b/iu.test(text.trim())
}

function decisionFor(text: string, category: MemoryCandidate["category"], explicit: boolean): MemoryCandidate["decision"] {
  if (explicit) return "save-approved"
  if (category === "project") return "save-approved"
  if (category === "preference") return "save-pending"
  return "discard"
}

function candidateFromText(text: string, explicit: boolean): MemoryCandidate | undefined {
  const normalized = normalizeMemoryText(text)
  if (!normalized || containsLikelySecret(normalized) || isQuestion(normalized)) return undefined
  const category = inferCategory(normalized)
  const scopeType = category === "project" ? "project" : "global"
  const decision = decisionFor(normalized, category, explicit)
  if (decision === "discard") return undefined
  return {
    text: normalized,
    category,
    scopeType,
    kind: inferMemoryKind(normalized, category),
    confidence: explicit ? 0.95 : 0.9,
    decision,
    reason: explicit ? "explicit memory request" : "durable user statement",
  }
}

function dedupe(candidates: MemoryCandidate[]): MemoryCandidate[] {
  const seen = new Set<string>()
  const result: MemoryCandidate[] = []
  for (const candidate of candidates) {
    const key = candidate.text.toLowerCase().replace(/\s+/gu, " ").trim()
    if (seen.has(key)) continue
    seen.add(key)
    result.push(candidate)
  }
  return result
}

export function extractStopCandidates(input: StopInput): MemoryCandidate[] {
  const user = input.lastUserMessage?.trim() ?? ""
  if (!user) return []

  const explicit = parseExplicitMemoryRequest(user)
  if (explicit) {
    const candidate = candidateFromText(explicit, true)
    return candidate ? [candidate] : []
  }

  const suggestion = detectUserMemorySuggestion(user)
  if (suggestion) {
    const candidate = candidateFromText(suggestion.text, false)
    return candidate ? [candidate] : []
  }

  return dedupe([candidateFromText(user, false)].filter((c): c is MemoryCandidate => Boolean(c)))
}
```

```ts
  if (input && typeof input === "object") {

      if (typeof obj[key] === "string") return obj[key] as string
    }
```

```ts
function persistCandidates(engine: MemoryEngine, candidates: MemoryCandidate[], input: StopInput | PostToolUseInput, event: "turn_stop" | "post_tool_use"): LifecycleResult {
  const saved: SaveResult[] = []
  const discarded: LifecycleResult["discarded"] = []
  for (const candidate of candidates) {
    if (candidate.decision === "discard") {
      discarded.push({ text: candidate.text, reason: candidate.reason })
      continue
    }
    const status = candidate.decision === "save-approved" ? "approved" : "pending"
    saved.push(engine.save({
      text: candidate.text,
      category: candidate.category,
      scopeType: candidate.scopeType,
      kind: candidate.kind,
      status,
      source: event === "turn_stop" && status === "approved" ? "user-suggested" : "agent-suggested",
      provenance: provenance(input, event, "toolName" in input ? input.toolName : undefined),
    }))
  }
  return { saved, discarded }
}
```

```ts
import type { PostToolUseInput, StopInput, UserPromptInput } from "@memory-lane/lifecycle"

export type CodexCommand = "user-prompt-submit" | "stop" | "post-tool-use"

export type ParsedCodexPayload =
  | { kind: "user-prompt-submit"; hookEventName: "UserPromptSubmit"; input: UserPromptInput }
  | { kind: "stop"; hookEventName: "Stop"; input: StopInput; transcriptPath?: string }
  | { kind: "post-tool-use"; hookEventName: "PostToolUse"; input: PostToolUseInput }
  | { kind: "invalid"; reason: string }

function stringField(obj: Record<string, unknown>, key: string): string | undefined {
  return typeof obj[key] === "string" ? obj[key] as string : undefined
}

function nullableStringField(obj: Record<string, unknown>, key: string): string | undefined {
  return typeof obj[key] === "string" ? obj[key] as string : undefined
}

export function parseCodexPayload(value: unknown): ParsedCodexPayload {
  if (!value || typeof value !== "object" || Array.isArray(value)) return { kind: "invalid", reason: "payload is not an object" }
  const obj = value as Record<string, unknown>
  const event = stringField(obj, "hook_event_name")
  const cwd = stringField(obj, "cwd")
  const sessionId = stringField(obj, "session_id")
  const turnId = stringField(obj, "turn_id")
  const model = stringField(obj, "model")
  const transcriptPath = nullableStringField(obj, "transcript_path")
  if (!event || !cwd) return { kind: "invalid", reason: "payload missing hook_event_name or cwd" }

  if (event === "UserPromptSubmit") {
    const prompt = stringField(obj, "prompt")
    if (prompt === undefined) return { kind: "invalid", reason: "UserPromptSubmit missing prompt" }
    return { kind: "user-prompt-submit", hookEventName: event, input: { cwd, sessionId, turnId, model, transcriptPath, prompt } }
  }

  if (event === "Stop") {
    const lastAssistantMessage = nullableStringField(obj, "last_assistant_message")
    return { kind: "stop", hookEventName: event, transcriptPath, input: { cwd, sessionId, turnId, model, transcriptPath, lastAssistantMessage } }
  }

  if (event === "PostToolUse") {
    const toolName = stringField(obj, "tool_name")
    if (!toolName) return { kind: "invalid", reason: "PostToolUse missing tool_name" }
    return {
      kind: "post-tool-use",
      hookEventName: event,
      input: {
        cwd,
        sessionId,
        turnId,
        model,
        transcriptPath,
        toolName,
        toolInput: obj.tool_input,
        toolResponse: obj.tool_response,
      },
    }
  }

  return { kind: "invalid", reason: `unsupported hook_event_name ${event}` }
}
```

```ts
import * as fs from "node:fs"

export interface TranscriptTurn {
  lastUserMessage?: string
  lastAssistantMessage?: string
}

const DEFAULT_MAX_BYTES = 200 * 1024

function contentToText(content: unknown): string | undefined {
  if (typeof content === "string") return content
  if (Array.isArray(content)) {
    const parts = content.map((part) => {
      if (typeof part === "string") return part
      if (part && typeof part === "object" && typeof (part as Record<string, unknown>).text === "string") return (part as Record<string, unknown>).text as string
      return ""
    }).filter(Boolean)
    return parts.length ? parts.join("\n") : undefined
  }
  return undefined
}

function roleFromObject(obj: Record<string, unknown>): string | undefined {
  const role = obj.role ?? obj.type ?? obj.author
  return typeof role === "string" ? role.toLowerCase() : undefined
}

export function readLatestTurnFromTranscript(filePath: string | undefined, maxBytes = DEFAULT_MAX_BYTES): TranscriptTurn {
  if (!filePath) return {}
  try {
    const stat = fs.statSync(filePath)
    const start = Math.max(0, stat.size - maxBytes)
    const fd = fs.openSync(filePath, "r")
    try {
      const buffer = Buffer.alloc(stat.size - start)
      fs.readSync(fd, buffer, 0, buffer.length, start)
      const lines = buffer.toString("utf8").split(/\r?\n/u).filter((line) => line.trim())
      let lastUserMessage: string | undefined
      let lastAssistantMessage: string | undefined
      for (const line of lines) {
        try {
          const parsed = JSON.parse(line) as Record<string, unknown>
          const role = roleFromObject(parsed)
          const text = contentToText(parsed.content ?? parsed.message ?? parsed.text)
          if (!role || !text) continue
          if (role.includes("user")) lastUserMessage = text
          if (role.includes("assistant")) lastAssistantMessage = text
        } catch { /* ignore malformed transcript lines */ }
      }
      return { lastUserMessage, lastAssistantMessage }
    } finally {
      fs.closeSync(fd)
    }
  } catch {
    return {}
  }
}
```

```ts
    return noopOutput("invalid JSON payload", options.env?.MEMORY_LANE_HOOK_DEBUG === "1")
  }

  const parsed = parseCodexPayload(payload)
  if (parsed.kind === "invalid") return noopOutput(parsed.reason, options.env?.MEMORY_LANE_HOOK_DEBUG === "1")
  if (parsed.kind !== command) return noopOutput(`event mismatch: command ${command} received ${parsed.kind}`, options.env?.MEMORY_LANE_HOOK_DEBUG === "1")

  try {
    if (parsed.kind === "user-prompt-submit") {
      return userPromptSubmitOutput(await handleUserPromptSubmit(options.engine, parsed.input), options.env?.MEMORY_LANE_HOOK_DEBUG === "1")
    }
    if (parsed.kind === "stop") {
      const latest = readLatestTurnFromTranscript(parsed.transcriptPath)
      const result = handleStop(options.engine, { ...parsed.input, ...latest })
      return lifecycleNoopOutput(result, options.env?.MEMORY_LANE_HOOK_DEBUG === "1")
    }
    const result = handlePostToolUse(options.engine, parsed.input)
    return lifecycleNoopOutput(result, options.env?.MEMORY_LANE_HOOK_DEBUG === "1")
  } catch {
    return noopOutput("hook handling failed", options.env?.MEMORY_LANE_HOOK_DEBUG === "1")
  }
}
```

`UserPromptSubmit` injects only approved memories that are relevant to the current prompt. It has strict item and character limits and injects nothing for generic prompts such as `ok`, `continue`, or `thanks`.

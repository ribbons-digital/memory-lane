# Codex SessionStart Baseline Injection Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `SessionStart` Codex hook that injects a small, recent, project-relevant baseline memory block at the start of every Codex session, before the first user prompt.

**Architecture:** Extend `@memory-lane/lifecycle` with a budgeted baseline selector and `handleSessionStart`, extend `@memory-lane/codex-adapter` to parse and run the new hook event, and wire `memory-lane codex session-start` into the CLI. `session_start` is already a valid `MemoryLifecycleEvent` in core, so no core storage changes are required.

**Tech Stack:** TypeScript ESM, Node.js stdlib, `node:test`, `tsx`, pnpm workspace, existing `@memory-lane/core` and `@memory-lane/lifecycle` APIs.

```ts
function compareBaselineRelevance(a: MemoryRecord, b: MemoryRecord): number {
  const dateCompare = b.updatedAt.localeCompare(a.updatedAt)
  if (dateCompare !== 0) return dateCompare
  // Tie-break: project-scoped memories before global memories
  const aProject = a.scope.type === "project" ? 1 : 0
  const bProject = b.scope.type === "project" ? 1 : 0
  return bProject - aProject
}

export function selectBaselineMemories(
  memories: MemoryRecord[],
  options?: Partial<MemoryInjectionLimits>,
): MemoryRecord[] {
  const limits = capLimits({ ...CODEX_BASELINE_INJECTION_LIMITS, ...options })
  const seen = new Set<string>()
  const selected: MemoryRecord[] = []
  let chars = 0

  const candidates = [...memories]
    .filter((m) => m.status === "approved" && !containsLikelySecret(m.text))
    .sort(compareBaselineRelevance)

  for (const memory of candidates) {
    if (selected.length >= limits.maxItems) break
    const key = normalizedMemoryKey(memory.text)
    if (!key || seen.has(key)) continue

    const remainingChars = limits.hardMaxChars - chars
    const fitted = fitMemoryWithinBudget(memory, remainingChars)
    if (!fitted) continue

    selected.push(fitted)
    seen.add(key)
    chars += fitted.text.length
  }

  return selected
}
```

```ts
export function handleSessionStart(
  engine: MemoryEngine,
  input: SessionStartInput,
  options?: Partial<MemoryInjectionLimits>,
): LifecycleResult {
  engine.refreshScope(input.cwd)
  const approved = engine.list({ status: "approved" })
  const selected = selectBaselineMemories(approved, options)
  const rendered = renderMemoryBlock(selected)
  return createResult(rendered || undefined)
}
```
